Table ​

Creates a table for displaying data.

<table class="c-table">
	<thead>
		<tr>
			<th>Head 1</th>
			<th>Head 2</th>
			<th>Head 3</th>
		</tr>
	</thead>
	<tbody>
		<tr>
			<td>Head 1</td>
			<td>Head 2</td>
			<td>Head 3</td>
		</tr>
		<tr>
			<td>Head 1</td>
			<td>Head 2</td>

		</tr>
	</tbody>
</table>
